**핵심 diff**(`index.html` `buildCollSets()`):

```js
const RELIC_CUT  = [3, 4, 3];
const RELIC_SETN = ['전사의 유물', '현자의 유물', '창세의 유물'];
let rcut = 0;
RELIC_CUT.forEach((c, si) => {
  const last = si === RELIC_CUT.length - 1;
  const part = last ? RELICS.slice(rcut) : RELICS.slice(rcut, rcut + c);   /* 종수가 늘면 마지막 세트로 */
  rcut += c;
  if(!part.length) return;
  add('relic:' + si, 'relic', 'relic', RELIC_SETN[si],
    part.reduce((n, r) => n + GRADE[r.g].mul, 0) / part.length, part.map(r => r.id));
});
```
